const NOVEMBER_2026_DAYS = 30

const calendarCells = Array.from({ length: 42 }, (_, i) => {
  const day = i - NOVEMBER_2026_FIRST_WEEKDAY + 1
  return day >= 1 && day <= NOVEMBER_2026_DAYS ? day : null
})
