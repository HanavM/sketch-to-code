import { Token, lex } from './lexer' import type { Block, Node } from './ast' // Parse one block: a heading line and its indented children. export function parseBlock(src: string, depth = 0): Block { const tokens: Token[] = lex(src) const children: Node[] = [] let i = 0 while (i < tokens.length) { const t = tokens[i++] if (t.kind === 'indent' && t.level > depth) { children.push(parseBlock(t.body, depth + 1)) } else if (t.kind === 'text') { children.push({ type: 'text', value: t.value }) } } return { type: 'block', depth, children } } export const MAX_DEPTH = 12
