-- revenue by country, last 30 days
SELECT c.country, count(*) AS orders,
sum(o.total) AS revenue
FROM orders o
JOIN customers c ON c.id = o.customer_id
WHERE o.created_at > now() - interval '30 days'
GROUP BY 1
ORDER BY revenue DESC
LIMIT 12;
